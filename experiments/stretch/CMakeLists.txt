cmake_minimum_required(VERSION 3.25)
project(VibeRaStretchExperiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(VIBERA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(STRETCH_ROOT "${VIBERA_ROOT}/vendor/source/tracktion_engine-4536d8a21664fe6ec2aa34b25abc87fa2a0d3b86/modules/tracktion_engine/3rd_party/signalsmith-stretch")
if(NOT EXISTS "${STRETCH_ROOT}/LICENSE.txt")
  message(FATAL_ERROR "Fetch the pinned engine source before this experiment")
endif()
add_executable(VibeRaStretchExperiment StretchExperiment.cpp)
target_include_directories(VibeRaStretchExperiment PRIVATE "${STRETCH_ROOT}")
if(MSVC)
  target_compile_options(VibeRaStretchExperiment PRIVATE /W4 /O2)
endif()
