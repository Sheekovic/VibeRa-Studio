cmake_minimum_required(VERSION 3.22)
project(VibeRaTuneExperiment LANGUAGES CXX)
add_executable(VibeRaTuneExperiment TuneExperiment.cpp)
get_filename_component(VIBERA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(STRETCH_ROOT "${VIBERA_ROOT}/vendor/source/tracktion_engine-4536d8a21664fe6ec2aa34b25abc87fa2a0d3b86/modules/tracktion_engine/3rd_party/signalsmith-stretch" CACHE PATH "Isolated experimental pitch-shifter candidate")
if(NOT EXISTS "${STRETCH_ROOT}/LICENSE.txt")
  message(FATAL_ERROR "Fetch the pinned engine source before the correction experiment")
endif()
add_executable(VibeRaTuneCorrectionExperiment TuneCorrectionExperiment.cpp)
set(TUNE_SIGNAL_PROFILE "Default" CACHE STRING "Signalsmith experimental window/hop profile")
if(TUNE_SIGNAL_PROFILE STREQUAL "Dense" OR TUNE_SIGNAL_PROFILE STREQUAL "DenseDouble")
  target_compile_definitions(VibeRaTuneCorrectionExperiment PRIVATE VIBERA_TUNE_WINDOW_MS=120 VIBERA_TUNE_HOP_MS=5)
elseif(TUNE_SIGNAL_PROFILE STREQUAL "Short")
  target_compile_definitions(VibeRaTuneCorrectionExperiment PRIVATE VIBERA_TUNE_WINDOW_MS=40 VIBERA_TUNE_HOP_MS=2)
elseif(NOT TUNE_SIGNAL_PROFILE STREQUAL "Default")
  message(FATAL_ERROR "Unknown experimental Signalsmith profile")
endif()
if(TUNE_SIGNAL_PROFILE STREQUAL "DenseDouble")
  target_compile_definitions(VibeRaTuneCorrectionExperiment PRIVATE VIBERA_TUNE_DOUBLE=1)
endif()
option(TUNE_FLUSH_DENORMALS "Experimental scoped x64 DAZ/FTZ comparison" OFF)
option(TUNE_CHANNELS_APART "Compare independent L/R with mid/side shifter processing" OFF)
option(TUNE_RESOURCE_AUDIT "Observe pinned Rubber Band C/aligned allocator entrypoints" OFF)
if(TUNE_RESOURCE_AUDIT)
  target_compile_definitions(VibeRaTuneCorrectionExperiment PRIVATE VIBERA_TUNE_RESOURCE_AUDIT=1)
endif()
if(TUNE_CHANNELS_APART)
  target_compile_definitions(VibeRaTuneCorrectionExperiment PRIVATE VIBERA_TUNE_CHANNELS_APART=1)
endif()
if(TUNE_FLUSH_DENORMALS)
  target_compile_definitions(VibeRaTuneCorrectionExperiment PRIVATE VIBERA_TUNE_FLUSH_DENORMALS=1)
endif()
target_include_directories(VibeRaTuneCorrectionExperiment PRIVATE "${STRETCH_ROOT}")
set(LINEAR_ROOT "" CACHE PATH "Optional isolated Linear include root")
if(LINEAR_ROOT)
  target_include_directories(VibeRaTuneCorrectionExperiment BEFORE PRIVATE "${LINEAR_ROOT}")
endif()
target_compile_features(VibeRaTuneCorrectionExperiment PRIVATE cxx_std_20)
set(RUBBERBAND_ROOT "" CACHE PATH "Optional isolated GPL Rubber Band candidate")
if(RUBBERBAND_ROOT)
  target_sources(VibeRaTuneCorrectionExperiment PRIVATE "${RUBBERBAND_ROOT}/single/RubberBandSingle.cpp")
  target_include_directories(VibeRaTuneCorrectionExperiment PRIVATE "${RUBBERBAND_ROOT}")
  target_compile_definitions(VibeRaTuneCorrectionExperiment PRIVATE VIBERA_TUNE_RUBBERBAND=1 NOMINMAX _USE_MATH_DEFINES)
endif()
target_compile_features(VibeRaTuneExperiment PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(VibeRaTuneExperiment PRIVATE /W4 /O2)
  target_compile_options(VibeRaTuneCorrectionExperiment PRIVATE /W4 /O2)
  target_compile_options(VibeRaTuneCorrectionExperiment PRIVATE /Zi)
  target_link_options(VibeRaTuneCorrectionExperiment PRIVATE /DEBUG:FULL)
endif()
if(WIN32)
  target_link_libraries(VibeRaTuneCorrectionExperiment PRIVATE dbghelp)
endif()
